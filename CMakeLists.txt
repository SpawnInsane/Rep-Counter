cmake_minimum_required(VERSION 3.20)
project(rep_counter VERSION 0.5.1 LANGUAGES CXX)

if(NOT WIN32)
    message(FATAL_ERROR "Rep Counter currently supports Windows only.")
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

set(REP_COUNTER_PRERELEASE "alpha")
set(REP_COUNTER_DISPLAY_VERSION "${PROJECT_VERSION}-${REP_COUNTER_PRERELEASE}")
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/cmake/version.hpp.in" "${CMAKE_CURRENT_BINARY_DIR}/generated/version.hpp" @ONLY)

add_library(rep_counter_core src/workout_session.cpp)
target_compile_features(rep_counter_core PUBLIC cxx_std_20)
target_include_directories(rep_counter_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_BINARY_DIR}/generated")

add_executable(rep_counter WIN32 src/main.cpp resources/rep-counter.rc)
target_compile_features(rep_counter PRIVATE cxx_std_20)
target_link_libraries(rep_counter PRIVATE rep_counter_core comctl32 dwmapi uxtheme winmm)
if(MSVC)
    target_link_options(rep_counter PRIVATE /MANIFEST:NO)
endif()

foreach(target rep_counter_core rep_counter)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()

include(CTest)
if(BUILD_TESTING)
    add_executable(rep_counter_tests tests/workout_session_tests.cpp)
    target_link_libraries(rep_counter_tests PRIVATE rep_counter_core)
    add_test(NAME workout_session COMMAND rep_counter_tests)
endif()

install(TARGETS rep_counter RUNTIME DESTINATION .)
install(FILES README.md RELEASE_NOTES.md LICENSE DESTINATION .)
set(CPACK_PACKAGE_NAME "Rep Counter")
set(CPACK_PACKAGE_VENDOR "Rep Counter contributors")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_FILE_NAME "Rep-Counter-${REP_COUNTER_DISPLAY_VERSION}-windows-x64")
set(CPACK_GENERATOR "ZIP")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
include(CPack)
